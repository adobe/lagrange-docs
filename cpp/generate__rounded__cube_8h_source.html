36struct RoundedCubeOptions : public PrimitiveOptions
37{
38 using Scalar = PrimitiveOptions::Scalar;
39 using Index = size_t;
40
42 Scalar width = 1;

45 Scalar height = 1;

51 Index width_segments = 1;

61 Scalar bevel_radius = 0;

65 Index bevel_segments = 8;
66
76 void project_to_valid_range()
77 {
78 width = std::max(width, Scalar(0));
79 height = std::max(height, Scalar(0));
80 depth = std::max(depth, Scalar(0));
81
82 width_segments = std::max(width_segments, Index(1));
83 height_segments = std::max(height_segments, Index(1));
84 depth_segments = std::max(depth_segments, Index(1));
85
86 bevel_radius = std::max(bevel_radius, Scalar(0));
87 Scalar max_acceptable_radius = (std::min(std::min(width, height), depth)) / 2;
88 bevel_radius = std::min(bevel_radius, max_acceptable_radius);
89
90 if (bevel_radius > this->epsilon) {
91 bevel_segments = std::max(bevel_segments, Index(1));
92 } else {
93 bevel_segments = 0;
94 }
95 }
96};
