38template <typename MeshType>
39std::unique_ptr<MeshType> generate_octahedron(
40 const typename MeshType::Scalar radius,
41 const Eigen::Matrix<typename MeshType::Scalar, 3, 1>& center = {0, 0, 0})
42{
43 using namespace Octahedron;
44
45 using VertexArray = typename MeshType::VertexArray;
46 using FacetArray = typename MeshType::FacetArray;
47 using Scalar = typename MeshType::Scalar;
48 using UVArray = typename MeshType::UVArray;
49
50 la_runtime_assert(radius >= 0.0, "Invalid radius: " + std::to_string(radius));
51
52 VertexArray vertices = generate_vertices<MeshType>(radius);
53 FacetArray facets = generate_facets<MeshType>();
54
55 // Translate to center
56 if (center.squaredNorm() > TOLERANCE<Scalar>) {
57 vertices.rowwise() += center.transpose();
58 }
59
60 auto mesh = create_mesh(vertices, facets);
61 UVArray uvs = compute_spherical_uv_mapping<MeshType>(*mesh, center);
62 mesh->add_corner_attribute("uv");
63 mesh->set_corner_attribute("uv", uvs);
64 map_corner_attribute_to_indexed_attribute(*mesh, "uv");
65 mesh->remove_corner_attribute("uv");
66
67 // Set uniform semantic label
68 lagrange::set_uniform_semantic_label(*mesh, PrimitiveSemanticLabel::SIDE);
69 return mesh;
70}

72namespace Octahedron {
73template <typename MeshType>
74typename MeshType::VertexArray generate_vertices(const typename MeshType::Scalar radius)
75{
76 typename MeshType::VertexArray vertices(6, 3);
77
78 vertices.row(0) << 0.0, -1.0, 0.0;
79 vertices.row(1) << 0.0, 0.0, 1.0;
80 vertices.row(2) << -1.0, 0.0, 0.0;
81 vertices.row(3) << 0.0, 0.0, -1.0;
82 vertices.row(4) << 1.0, 0.0, 0.0;
83 vertices.row(5) << 0.0, 1.0, 0.0;
84
85 return radius * vertices;
86}
87
88template <typename MeshType>
89typename MeshType::FacetArray generate_facets()
90{
91 typename MeshType::FacetArray facets(8, 3);
92
93 facets << 0, 1, 2, 0, 2, 3, 0, 3, 4, 0, 4, 1,
94
95 5, 2, 1, 5, 3, 2, 5, 4, 3, 5, 1, 4;
96
97 return facets;
98}
