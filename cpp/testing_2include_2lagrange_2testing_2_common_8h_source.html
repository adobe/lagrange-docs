166template <typename Scalar, typename Index>
167SurfaceMesh<Scalar, Index> load_surface_mesh(const fs::path& relative_path)
168{
169 auto full_path = get_data_path(relative_path);
170 REQUIRE(lagrange::fs::exists(full_path));
171 io::LoadOptions options;
172 options.quiet = true;
173 return lagrange::io::load_mesh<SurfaceMesh<Scalar, Index>>(full_path, options);
174}
