24template <typename ValueType>
25constexpr char pcd_type_char()
26{
27 if constexpr (std::is_floating_point_v<ValueType>) {
28 return 'F';
29 } else if constexpr (std::is_signed_v<ValueType>) {
30 return 'I';
31 } else {
32 return 'U';
33 }
34}

45template <typename Func>
46void dispatch_pcd_type(char type, int size, Func&& func)
47{
48 switch (type) {
49 case 'F':
50 if (size == 4) return void(func(float{}));
51 if (size == 8) return void(func(double{}));
52 break;
53 case 'I':
54 if (size == 1) return void(func(int8_t{}));
55 if (size == 2) return void(func(int16_t{}));
56 if (size == 4) return void(func(int32_t{}));
57 if (size == 8) return void(func(int64_t{}));
58 break;
59 case 'U':
60 if (size == 1) return void(func(uint8_t{}));
61 if (size == 2) return void(func(uint16_t{}));
62 if (size == 4) return void(func(uint32_t{}));
63 if (size == 8) return void(func(uint64_t{}));
64 break;
65 default: break;
66 }
67 throw Error(lagrange::format("Unsupported PCD field type '{}' with size {}", type, size));
68}
